Algoritmo ventas_prodcutos3_funciones
	
	definir u1, p1, s1 como real
	definir u2, p2, s2 Como Real
	definir u3, p3, s3 Como Real
	
	u1 <- pedir_unidades ("prodcuto 1")
	p1 <- pedir_precio ("producto 1")
	s1 <- calcular_subtotal (u1, p1)
	
	u2 <- pedir_unidades ("prodcuto 2")
	p2 <- pedir_precio ("producto 2")
	s2 <- calcular_subtotal (u1, p1)
	
	u3 <- pedir_unidades ("prodcuto 3")
